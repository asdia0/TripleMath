\section{Area}\label{sec:Applications-of-Integration:Area}

\subsection{The Riemann Sum and Integral}

Suppose we wish to find exact area bounded by the graph of $y = f(x)$, the $x$-axis and the lines $x = a$ and $x = b$, where $a \leq b$ and $f(x) \geq 0$ for $a \leq x \leq b$. 

We can approximate this area by drawing $n$ rectangles of equal width, as shown in the figure below:

Observe that the $k$th rectangle has width $\D x = (b-a)/n$ and height $f(a + k \D x)$. The total area of the rectangles is hence \[\sum_{k = 1}^n f(a + k \D x) \D x.\] This is known as the \vocab{Riemann sum} of $f$ over $[a, b]$.

As the number of rectangles increases, the width $\D x$ of the rectangles approaches 0, and the total area of rectangles approaches the actual area under the curve. In other words, \[\area = \lim_{\D x \to 0} \sum_{k = 1}^n f(a + k \D x) \D x.\] In the limit, the Riemann sum becomes the \vocab{Riemann integral}, which is conventionally written as the definite integral \[\int_a^b f(x) \d x.\]

Note that this is where the notation for integration comes from: in the limit, $\sum \to \int$ and $\D x \to \d x$.

\subsection{Cartesian Form}

\begin{proposition}[Area between a Curve and the $x$-axis]
    The region bounded by the curve $y = f(x)$, the $x$-axis, and the lines $x = a$ and $x = b$ has area \[\int_a^b \abs{y} \d x = \int_a^b \abs{f(x)} \d x.\]
\end{proposition}

More generally, we have the following result.

\begin{proposition}[Area between Two Curves with respect to the $x$-axis]
    The region bounded by the curves $y = f(x)$ and $y = g(x)$, and the lines $x = a$ and $x = b$ has area \[\int_a^b \abs{f(x) - g(x)} \d x.\]
\end{proposition}

Similar results hold when integrating with respect to the $y$-axis instead.

\begin{proposition}[Area between a Curve and the $y$-axis]
    The region bounded by the curve $x = f(y)$, the $y$-axis, and the lines $y = a$ and $y = b$ has area \[\int_a^b \abs{x} \d y = \int_a^b \abs{g(y))} \d y.\]
\end{proposition}

\begin{proposition}[Area between Two Curves with respect to the $y$-axis]
    The region bounded by the curves $x = f(y)$ and $x = g(y)$, and the lines $y = a$ and $y = b$ has area \[\int_a^b \abs{f(y) - g(y)} \d y.\]
\end{proposition}

\subsection{Parametric Form}

\begin{proposition}[Area between a Parametric Curve and the $x$-axis]
    Let $C$ be the curve with parametric equations $x = f(t)$ and $y = g(t)$. Then the region bounded by $C$, the $x$-axis, and the lines $x = a$ and $x = b$ has area \[\int_a^b \abs{y} \d x = \int_{t_1}^{t_2} \abs{g(t)} \der{x}{t} \d t,\] where $t_1$ and $t_2$ are the values of $t$ when $x = a$ and $b$ respectively.
\end{proposition}

An analogous formula holds for the area between $C$ and the $y$-axis.

\begin{proposition}[Area between a Parametric Curve and the $y$-axis]
    Let $C$ be the curve with parametric equations $x = f(t)$ and $y = g(t)$. Then the region bounded by $C$, the $y$-axis, and the lines $y = a$ and $y = b$ has area \[\int_a^b \abs{x} \d y = \int_{t_1}^{t_2} \abs{f(t)} \der{y}{t} \d t,\] where $t_1$ and $t_2$ are the values of $t$ when $y = a$ and $b$ respectively.
\end{proposition}

\subsection{Polar Form}

\begin{proposition}[Area Enclosed by Polar Curve]
    Let $A$ be the region bounded by the polar curve $r = f(\t)$ curve and two half-lines $\t = \a$ and $\t = \b$. Then $A$ has area \[\frac12 \int_{\a}^{\b} r^2 \d \t.\]
\end{proposition}

\begin{proof}
    Divide the enclosed region $A$ into $n$ sectors with the same interior angle $\D \t$. The area of each sector can be approximated by a sector of a circle, so \[\D A \approx \frac12 r^2 \D \t.\] Summing up these approximations, we see that \[A \approx \sum_{\t = \a}^{\t = \b} \frac12 r^2 \D \t.\] This approximation will improve as the number of sectors increases, i.e. $\D \t \to 0$. Hence, \[\area A = \lim_{\D \t \to 0} \sum_{\t = \a}^{\t = \b} \frac12 r^2 \D \t = \frac12 \int_{\a}^{\b} r^2 \d \t.\]
\end{proof}

\section{Volume}

\begin{definition}
    If an enclosed region is rotated about a straight line, the three-dimensional object formed is called a \vocab{solid of revolution}, and its volume is a \vocab{volume of revolution}.
\end{definition}

The line about which rotation takes place is always an axis of symmetry for the solid of revolution, and any cross-section of the solid which is perpendicular to the axis of rotation is circular.

\subsection{Disc Method}

\begin{proposition}[Disc Method about $x$-axis]
    When the region bound by the curve $y = f(x)$, the $x$-axis and the lines $x = a$ and $x = b$ is rotated $2\pi$ radians about the $x$-axis, the volume of the solid of revolution generated is given by \[V = \pi \int_a^b y^2 \d x = \pi \int_a^b \bs{f(x)}^2 \d x.\]
\end{proposition}

\begin{proof}
    To calculate the volume of the solid of revolution, we can cut it into thin slices (or discs) of thickness $\D x$. Each disc is approximately a cylinder and the approximate volume of the solid can be found by summing the volumes of these cylinders. The smaller $\D x$ is, the better the approximation.

    Consider a typical disc formed by a one cut through the point $P(x, y)$ and the other cut distant $\D x$ from the first. The volume of this disc is approximately \[\D V \approx \pi y^2 \D x.\] Summing over all discs, \[V \approx \sum_{x = a}^{b} \pi y^2 \D x.\] As more cuts are made, $\D x \to 0$, whence \[V = \lim_{\D x \to 0} \sum_{x = a}^{b} \pi y^2 \D x = \pi \int_a^b y^2 \d x.\]
\end{proof}

A similar result holds when the axis of rotation is the $y$-axis.

\begin{proposition}[Disc Method about $y$-axis]
    When the region bound by the curve $x = g(y)$, the $y$-axis and the lines $y = a$ and $y = b$ is rotated $2\pi$ radians about the $y$-axis, the volume of the solid of revolution generated is given by \[V = \pi \int_a^b x^2 \d y = \pi \int_a^b \bs{g(y)}^2 \d y.\]
\end{proposition}

\subsection{Shell Method}

\begin{proposition}[Shell Method about $y$-axis]
    When the region bound by the curve $y = f(x)$, the $x$-axis and the lines $x = a$ and $x = b$ is rotated $2\pi$ radians about the $y$-axis, the volume of the solid of revolution is given by \[V = 2\pi \int_a^b xy \d x.\]
\end{proposition}

    \end{tikzpicture}

\begin{proof}
    Consider a typical vertical strip in the bounded region with height $y$ and thickness $\D x$. When rotated about the $y$-axis, it will form a cylindrical shell with inner radius $x$, outer radius $x + \D x$ and height $y$, contributing a volume of
    \begin{align*}
        \D V &= \pi (x + \D x)^2 y - \pi x^2 y\\
        &= 2 \pi xy \D x + \pi \D x^2 y\\
        &\approx 2 \pi xy \D x.
    \end{align*}
    Summing all of these contributions, the volume of revolution is approximated by \[V \approx \sum_{x = a}^b 2 \pi xy \D x.\] As more strips are considered, $\D x \to 0$, whence \[V = \lim_{\D x \to 0} \sum_{x = a}^b 2 \pi xy \D x = 2\pi \int_a^b xy \d x.\]
\end{proof}

A similar result holds when the axis of rotation is the $x$-axis.

\begin{proposition}[Shell Method about $x$-axis]
    When the region bound by the curve $x = g(y)$, the $y$-axis and the lines $y = a$ and $y = b$ is rotated $2\pi$ radians about the $x$-axis, the volume of the solid of revolution is given by \[V = 2\pi \int_a^b xy \d y.\]
\end{proposition}

\section{Arc Length}

\subsection{Parametric Form}

\begin{proposition}[Arc Length of Parametric Curve]
    Let $A(t_1)$ and $B(t_2)$ be points on the parametric curve with equations $x = f(t)$ and $y = g(t)$. The arc length of the curve from $A$ to $B$ is \[\int_{t_1}^{t_2} \sqrt{\bs{f'(t)}^2 + \bs{g'(t)}^2} \d t.\]
\end{proposition}

\begin{proof}
    Let $s$ be the arc length of the curve from $A$ to $B$. Consider splitting $AB$ into many tiny curve segments, where each endpoint has parameters differing by $\D t$. Consider one such curve segment $PQ$, where $P$ and $Q$ have parameters $t$ and $t + \D t$ respectively. The arc length from $P$ to $Q$ can be approximated by the straight line distance between $P$ and $Q$. By the Pythagorean theorem, we have \[\D s \approx \sqrt{\bs{f(t + \D t) - f(t)}^2 + \bs{g(t + \D t) - g(t)}^2}.\] Summing $\D s$ over each tiny curve segment, the arc length from $A$ to $B$ is approximated by \[s \approx \sum_{t = t_1}^{t_2} \sqrt{\bs{f(t + \D t) - f(t)}^2 + \bs{g(t + \D t) - g(t)}^2}.\] Factorizing a $\D t$ out of the surd, we see that \[s \approx \sum_{t = t_1}^{t_2} \sqrt{\bs{\frac{f(t + \D t) - f(t)}{\D t}}^2 + \bs{\frac{g(t + \D t) - g(t)}{\D t}}^2} \D t.\] As more and more curve segments are considered, $\D t \to 0$, whence
    \begin{align*}
        s &= \lim_{\D t \to 0} \sum_{t = t_1}^{t_2} \sqrt{\bs{\frac{f(t + \D t) - f(t)}{\D t}}^2 + \bs{\frac{g(t + \D t) - g(t)}{\D t}}^2} \D t\\
        &= \int_{t_1}^{t_2} \sqrt{\bs{f'(t)}^2 + \bs{g'(t)}^2} \d t.
    \end{align*}
\end{proof}

\subsection{Cartesian Form}

Taking $t = x$ or $t = y$, we get the following formulas involving $\derx{y}{x}$ and $\derx{x}{y}$, which is suitable for Cartesian curves.

\begin{proposition}[Arc Length of Cartesian Curve]
    Let $A(x_1, y_1)$ and $B(x_2, y_2)$ lie on the Cartesian curve $y = f(x)$ or $x = g(y)$. The arc length of the curve from $A$ to $B$ is given by \[\int_{x_1}^{x_2} \sqrt{1 + \bs{f'(x)}^2} \d x = \int_{y_1}^{y_2} \sqrt{\bs{g'(y)}^2 + 1} \d y.\]
\end{proposition}

\subsection{Polar Form}

\begin{proposition}
    Let $A(r_1, \t_1)$ and $B(r_2, \t_2)$ be points on the polar curve $r = f(\t)$. Then the arc length of the curve from $A$ to $B$ is given by \[\int_{\t_1}^{\t_2} \sqrt{\bs{r(\t)}^2 + \bs{r'(\t)}^2} \d \t.\]
\end{proposition}
\begin{proof}
    Recall that $x = r\cos \t$ and $y = r\sin \t$. Hence, \[\der{x}{\t} = \cos \t \der{r}{\t} - r \sin \t \quad \tand \quad \der{y}{\t} = \sin \t \der{r}{t} + r \cos \t.\] It follows that \[\bp{\der{x}{\t}}^2 + \bp{\der{y}{\t}}^2 = \bp{\cos^2 \t + \sin^2 \t} \bs{r^2 + \bp{\der{r}{\t}}^2} = r^2 + \bp{\der{r}{\t}}^2.\] Taking $t = \t$, the arc length from $A$ to $B$ is \[\int_{\t_1}^{\t_2} \sqrt{\bp{\der{x}{\t}}^2 + \bp{\der{y}{\t}}^2} \d \t = \int_{\t_1}^{\t_2} \sqrt{r^2 + \bp{\der{r}{\t}}^2} \d \t.\]
\end{proof}

\section{Surface Area of Revolution}

\begin{definition}
    The surface area of a solid of revolution is called the \vocab{surface area of revolution}.
\end{definition}

\begin{proposition}[Surface Area of Revolution of Parametric Curve]
    Let $A(t_1)$ and $B(t_2)$ be points on the parametric curve with equations $x = f(t)$ and $y = g(t)$. The surface area of revolution about the $x$-axis of arc $AB$ is given by \[A_x = 2\pi \int_{t_1}^{t_2} y \sqrt{\bs{f'(t)}^2 + \bs{g'(t)}^2} \d t,\] while the surface area of revolution about the $y$-axis is given by \[A_y = 2\pi \int_{t_1}^{t_2} x \sqrt{\bs{f'(t)}^2 + \bs{g'(t)}^2} \d t.\]
\end{proposition}
We prove only the formula for a revolution about the $x$-axis. The proof of the $y$-axis case is identical.

            \fill (1.2, 2.44) circle[radius=2.5pt];

            \draw[->] (0.8, -1.22) -- (1.04, -1.22);
            \draw[->] (1.64, -1.22) -- (1.4, -1.22);
            \node[anchor=west] at (1.63, -1.22) {$\D x$};
        \end{axis}
    \end{tikzpicture}
    \caption{}
\end{figure}
\begin{proof}
    Let $s$ be the arc length of the curve from $A$ to $B$. Consider splitting $AB$ into many tiny curve segments, where each endpoint has parameters differing by $\D t$. Consider one such curve segment $PQ$, where $P$ and $Q$ have parameters $t$ and $t + \D t$ respectively. Recall that the arc length between $P$ and $Q$ is approximately \[\D s \approx \sqrt{\bs{f(t + \D t) - f(t)}^2 + \bs{g(t + \D t) - g(t)}^2}.\]

    Now consider the surface area of revolution about the $x$-axis of arc $PQ$. For small $\D s$, the solid of revolution is approximately a disc with radius $y$ and width $\D s$. The surface area of this disc can be approximated by \[\D A_x \approx 2 \pi y \D s = 2 \pi y \sqrt{\bs{f(t + \D t) - f(t)}^2 + \bs{g(t + \D t) - g(t)}^2}.\]

    Summing $\D A_x$ over each curve segment, $A_x$ is approximated by
    \begin{align*}
        A_x &\approx \sum_{t = t_1}^{t_2} 2 \pi y \sqrt{\bs{f(t + \D t) - f(t)}^2 + \bs{g(t + \D t) - g(t)}^2}\\
        &= \sum_{t = t_1}^{t_2} 2 \pi y \sqrt{\bs{\frac{f(t + \D t) - f(t)}{\D t}}^2 + \bs{\frac{g(t + \D t) - g(t)}{\D t}}^2} \D t.
    \end{align*}
    As more and more curve segments are considered, $\D t \to 0$, whence
    \begin{align*}
        A_x &= \lim_{\D t \to 0} 2 \pi y \sqrt{\bs{\frac{f(t + \D t) - f(t)}{\D t}}^2 + \bs{\frac{g(t + \D t) - g(t)}{\D t}}^2} \D t\\
        &= \int_{t_1}^{t_2} 2\pi y \sqrt{\bs{f'(t)}^2 + \bs{g'(t)}^2} \d t.
    \end{align*}
\end{proof}

\section{Approximating Definite Integrals}

In \SS\ref{sec:Applications-of-Integration:Area}, we saw how Riemann sums could approximate definite integrals using rectangles. This is a blunt tool which utilizes very little information from the curve and thus will often not give a good estimate of the definite integral for a fixed number of rectangles.

In this chapter, we will be exploring two other methods: the trapezium rule and Simpson's rule, for finding the approximate value of an area under a curve. These methods often give better approximations to the actual area as compared to using Riemann sums. Similar to Riemann sums, these methods can be extended to estimate the value of a definite integral.

\subsection{Trapezium Rule}

Consider the curve $y = f(x)$ which is non-negative over the interval $[a, b]$.

Divide the interval $[a, b]$ into $n$ equal intervals (strips), each with width $h = (b-a)/n$. Letting $y_i = a + ih$, we see that the area of the $n$ trapeziums is given by \[\area = \sum_{k = 0}^n \frac{h}{2} \bp{y_k + y_{k+1}} = \frac{h}{2} \bs{y_0 + 2(y_1 + y_2 + \dots + y_{n-1}) + y_n}.\]

\begin{recipe}[Trapezium Rule]
    The trapezium rule with ($n+1$) ordinates (or $n$ intervals) gives the approximation \[\int_a^b f(x) \d x \approx \sum_{k = 0}^n \frac{h}{2} \bp{y_k + y_{k+1}} = \frac{h}{2} \bs{y_0 + 2(y_1 + y_2 + \dots + y_{n-1}) + y_n},\] where $h = (b-a)/n$.
\end{recipe}

\begin{sample}\label{samp:Trapezium-Rule}
    Use the trapezium rule with 4 strips to find an approximation for \[\int_0^2 \ln{x + 2} \d x.\] Find the percentage error of the approximation.
\end{sample}
\begin{sampans}
    Let $f(x) = \ln{x+2}$. By the trapezium rule,
    \begin{align*}
        \int_0^2 \ln{x + 2} \d x &\approx \frac{1}{2} \cdot \frac{2 - 0}{4} \Big(f(0) + 2\bs{f(0.5) + f(1) + f(1.5)} + f(2)\Big) \\
        &= 2.15369 \todp{5}.
    \end{align*}
    One can easily verify that the integral evaluates to $2.15888 \todp{5}$. Hence, the percentage error is \[\abs{\frac{2.15888 - 2.15369}{2.15888}} = 0.240 \%.\]
\end{sampans}

\subsubsection{Error in Trapezium Rule Approximation}

The errors in the trapezium rule are influenced by the concavity of the curve.
\begin{itemize}
    \item If the curve is concave upward, the secant lines lie above the curve. Hence, the trapezium rule will give an overestimate.
    \item If the curve is concave downward, the secant lines lie below the curve. Hence, the trapezium rule will give an underestimate.
\end{itemize}

\subsection{Simpson's Rule}

Previously, we explored how Riemann sums approximate definite integrals using horizontal lines (i.e. degree 0 polynomials). We also saw how the trapezium rule improves this approximation by using sloped lines (i.e. degree 1 polynomials). Now, we introduce Simpson's rule, which takes this a step further by using quadratics (i.e. degree 2 polynomials) to achieve even greater accuracy in approximating definite integrals.

Consider the curve $y = f(x)$, which is non-negative over the interval $[a, b]$. Suppose the area represented by $\int_a^b f(x) \d x$ is divided by the ordinates $y_0$, $y_1$, $y_2$ into two strips each of width $h$ as shown below. A particular parabola can be found passing through the three points on the curve with ordinates $y_0$, $y_1$, $y_2$. Simpson's rule uses the area under the parabola to approximate the area represented by $\int_a^b f(x) \d x$.

To deduce the area under the parabola, we consider the case where $y = f(x)$ is translated $x_1$ units to the left, i.e. the line $x = x_1$ is now the $y$-axis.

Under this translation, \[\int_a^b f(x) \d x = \int_{-h}^h f(x + x_1) \d x.\] This area will now be approximated by a parabola $y = g(x) = Ax^2 + Bx + C$, where $A$, $B$ and $C$ are constants. Since the parabola intersects the curve at $(-h, y_1)$, $(0, y_2)$ and $(h, y_3)$, we have \[g(-h) = Ah^2 - Bh + C = y_0, \quad g(0) = C = y_1, \quad g(h) = Ah^2 + Bh + C = y_2.\]

The area under the parabola is thus
\begin{align*}
    \int_{-h}^h \bp{Ax^2 + Bx + C} \d x &= \frac{h}{3} \bp{2Ah^2 + 6C}\\
    &= \frac{h}{3} \bs{\bp{Ah^2 - Bh + C} + 4C + \bp{Ah^2 + Bh + C}}\\
    &= \frac{h}{3} (y_0 + 4y_1 + y_2).
\end{align*}

We hence arrive at Simpson's rule with 2 strips: \[\int_a^b f(x) \d x \approx \frac{h}{3} (y_0 + 4y_1 + y_2).\] We can extend Simpson's rule to cover any even number of strips. In general,

\begin{recipe}[Simpson's Rule]
    Simpson's rule with $2n$ strips (or $2n+1$ ordinates) gives the approximation
    \begin{align*}
        \int_a^b f(x) \d x &\approx \sum_{k = 0}^n \frac{h}{3} (y_{2k} + 4y_{2k+1} + y_{2k+2})\\
        &= \frac{h}{3} \bs{y_0 + 4y_1 + 2y_2 + 4y_3 + 2y_4 + \dots + 2y_{2n-2} + 4y_{2n-1} + y_{2n}}.
    \end{align*}
\end{recipe}

\begin{sample}
    Use Simpson's rule with 4 strips to find an approximation for \[\int_0^2 \ln{x + 2} \d x.\] Find the percentage error of the approximation.
\end{sample}
\begin{sampans}
    Let $f(x) = \ln{x+2}$. By the trapezium rule,
    \begin{align*}
        \int_0^2 \ln{x + 2} \d x &\approx \frac{1}{3} \cdot \frac{2 - 0}{4} \Big[f(0) + 4f(0.5) + 2f(1) + 4f(1.5) + f(2)\Big] \\
        &= 2.15881 \todp{5}.
    \end{align*}
    As previously mentioned in Sample Problem~\ref{samp:Trapezium-Rule} the actual value of the integral is $2.15888 \todp{5}$. Hence, the percentage error is \[\abs{\frac{2.15888 - 2.15881}{2.15888}} = 0.00324\%.\]
\end{sampans}

In the previous example, the trapezium rule gave an estimate of $2.15369 \todp{5}$, which has an error of 0.240\%. In the case of Simpson's rule, the error is 0.00324\%, vastly better than that of the trapezium rule's.

In general, Simpson's rule gives a better approximation than the trapezium rule as the quadratics used account for the concavity of the curve.
