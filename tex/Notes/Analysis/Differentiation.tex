\chapter{Differentiation}

\section{First Principles}

\begin{definition}
    A function $f$ is said to be \vocab{differentiable} at some point $a$, if its domain contains an open interval containing $a$, and the limit \[L = \lim_{h \to 0} \frac{f(a+h) - f(a)}{h}\] exists. If $f$ is differentiable, then $L$ is the \vocab{derivative} of $f$ at $a$, denoted by $\frac{\d}{\d x} f(a)$, or $f'(a)$.
    
    If the derivative exists for all points in the domain, we may define a \vocab{derivative function} that maps $x$ to the value of the derivative at $x$, denoted by $\frac{\d}{\d x} f(x)$, $\derx{f}{x}$, or $f'(x)$. If $y = f(x)$, we write the derivative as $\frac{\d y}{\d x}$, $\derx{y}{x}$, or $y'(x)$.
\end{definition}


The derivative of $f$ at $a$ can be understood as the instantaneous rate of change of $f(x)$ at $a$. Consider two points $A(a, f(a))$ and $B(a + h, f(a + h))$ on the curve $y = f(x)$.

The gradient of the chord $AB$, denoted $m_{AB}$ is given by \[m_{AB} = \frac{f(a + h) - f(a)}{(a + h) - a} = \frac{f(a + h) - f(a)}{h}.\] From the above figure, one can see that the closer $B$ is to $A$, the better $m_{AB}$ approximates the gradient of the curve at $A$. Formally, \[\text{gradient at A} = \lim_{B \to A} m_{AB} = \lim_{h \to 0} \frac{f(a + h) - f(a)}{(a + h) - a} = \frac{f(a + h) - f(a)}{h}.\] This is the definition of the derivative of $f$ at $a$!

\begin{definition}
    For positive integers $n$, the \vocab{$n$th derivative} of $y$ with respect to $x$ is \[\der[n]{y}{x} = f^{(n)}(x) = \der{}{x} \bp{\der[n-1]{y}{x}}.\]
\end{definition}

\section{Differentiation Rules}

\begin{proposition}[Differentiation Rules]
    Let $k \in \RR$ and suppose $u$ and $v$ are functions of $x$. Then
    \begin{itemize}
        \item (Sum/Difference Rule) If $y = u \pm v$ then $y' = u' \pm v'$.
        \item (Product Rule) If $y = uv$, then $y' = u' v + u v'$.
        \item (Quotient Rule) If $y = u/v$, then \[y' = \frac{u' v - u v'}{v^2}.\]
        \item (Chain Rule) If $y = f(x)$ and $x = g(t)$, then \[\der{y}{t} = \der{y}{x} \der{x}{t}.\]
    \end{itemize}
\end{proposition}
The sum, product and quotient rules are easy to prove from first principles. We hence only prove the chain rule. To do so, we introduce an equivalent definition of differentiability at a point.
\begin{definition}\label{def:Differentiability}
    A function $f(x)$ is \vocab{differentiable} at $a$ if there exists some function $q(x)$ continuous at $a$ such that \[q(x) = \frac{f(x) - f(a)}{x - a}.\] Note that there is at most one such $q(x)$, and if it exists, then $q(x) = f'(x)$.
\end{definition}
We now prove the chain rule.
\begin{proof}[Proof of Chain Rule]    
    Suppose $y = f(x)$ and $x = g(t)$. Suppose also that $f(x)$ is differentiable at $x = g(a)$, and that $g(t)$ is differentiable at $a$.

    Since $f(x)$ is differentiable at $x = g(a)$, by the above definition, there exists a function $q(x)$ such that \[q(x) = \frac{f(x) - f(g(a))}{x - g(a)}.\] Replacing $x$ with $g(t)$, and rearranging terms, we get \[g(t) - g(a) = \frac{f(g(t)) - f(g(a))}{q(g(t))}. \tag{$\ast$}\]

    Similarly, since $g(t)$ is differentiable at $a$, by the above definition, there must exist a function $r(t)$ continuous at $a$ such that \[r(t) = \frac{g(t) - g(a)}{t - a}.\] This implies that $g(t) - g(a) = r(t) (t - a)$. Equating this with ($\ast$), we have \[\frac{f(g(t)) - f(g(a))}{q(g(t))} = r(t) (t - a).\] Rearranging, \[q(g(t)) r(t) = \frac{f(g(t)) - f(g(a))}{t - a} = \frac{(f \circ g)(t) - (f \circ g)(a)}{t - a}.\] By our assumptions, $q(g(t)) r(t)$ is continuous at $t = a$. Hence, by the above definition, $q(g(t)) r(t)$ is the derivative of $(f \circ g)'(t)$. Since $q(x) = f'(x)$ and $r(t) = g'(t)$, we arrive at $(f \circ g)'(t) = f'(g(t)) g'(t)$. Since $x = g(t)$ and $y = f(x) = f(g(t))$, this can be written more compactly as \[\der{y}{t} = \der{y}{x} \der{x}{t}.\]
\end{proof}

From the chain rule, we can derive the following property:

\begin{proposition}
    Suppose $\derx{x}{y} \neq 0$. Then \[\der{y}{x} = \frac1{\derx{x}{y}}.\]
\end{proposition}
\begin{proof}
    By the chain rule, \[1 = \der{y}{y} = \der{y}{x} \der{x}{y}.\] Since $\derx{x}{y} \neq 0$, we may divide by $\derx{x}{y}$ to get \[\der{y}{x} = \frac1{\derx{x}{y}}.\]
\end{proof}

Note that this property does not generalize to higher derivatives. For example, \[\der[2]{y}{x} \neq \frac1{\derx[2]{x}{y}}\] in general.

\section{Implicit Differentiation}

\begin{definition}
    An \vocab{explicit function} is one of the form $y = f(x)$, i.e. the dependent variable $y$ is expressed explicitly in terms of the independent variable $x$. An \vocab{implicit function} is a function defined by an equation $f(x, y) = 0$.
\end{definition}

\begin{example}
    $y = 2x \sin x + 3$ is an explicit function, while $xy + \sin y = 2$ defines an implicit function $y$.
\end{example}

\begin{recipe}[Implicit Differentiation]
    Given an implicit function $y$ defined by $f(x, y) = 0$, where $y$ is regarded as a function of $x$, we can find $y'$ by differentiating both sides of the equation with respect to $x$, applying the chain rule and rearranging.
\end{recipe}

\begin{example}[Implicit Differentiation]
    Let $y$ be defined by the equation $3y^3 = 2 - x^2 y$. Implicitly differentiating the equation with respect to $x$, we obtain $9y^2 y' = -x^2 y' - 2x y$. Solving for $y'$, we get \[y' = \frac{-2xy}{9y^2 + x^2}.\]
\end{example}

\begin{proposition}[Derivative of Inverse Functions]
    If $f$ has an inverse, its derivative is given by \[\der{}{x} \inv f(x) = \frac1{{f'\bp{\inv f(x)}}}.\]
\end{proposition}
\begin{proof}
    Let $y = \inv f(x)$, so $f(y) = x$. Implicitly differentiating, we have $f'(y) \, y' = 1$, so \[y' = \frac1{f'(y)} = \frac1{{f'\bp{\inv f(x)}}}.\]
\end{proof}

We can use the above result to derive the derivatives of the inverse trigonometric functions and the logarithm.

\section{Parametric Differentiation}

Suppose $x = f(t)$ and $y = g(t)$ are given in terms of a parameter $t$. Sometimes it is difficult to obtain the Cartesian form of a parametric equation, so we may not be able to write $\derx{y}{x}$ in terms of $x$ and $y$. However, we can write $\derx{y}{x}$ in terms of the parameter $t$ using the chain rule: \[\der{y}{x} = \der{y}{t} \der{t}{x} = \frac{\derx{y}{t}}{\derx{x}{t}}.\]

\begin{example}[Parametric Differentiation]
    Suppose $x = \sin 2\t$, $y = \cos 4\t$. Differentiating $x$ and $y$ with respect to $\t$, we see that \[\der{x}{\t} = 2\cos 2\t \quad \tand \quad \der{y}{\t} = -4\sin 4\t.\] Thus, by the chain rule, \[\der{y}{x} = \der{y}{\t} \der{\t}{x} = \frac{-2 \sin 4\t}{\cos 2\t}.\]
\end{example}
